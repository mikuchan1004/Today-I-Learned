package var_02;

import java.util.ArrayList;
import java.util.List;

/**
 * ============================================================================
 * [List_08.java - 컬렉션 프레임워크: ArrayList & List 다루기]
 * ============================================================================
 * 
 * 1. 컬렉션 프레임워크(Collection Framework)란?
 *    - 다수의 데이터를 쉽고 효율적으로 다룰 수 있도록 제공되는 자바의 표준 라이브러리입니다.
 *    - 일반 배열(Array)은 크기가 한 번 정해지면 바꿀 수 없지만, 
 *      ArrayList와 같은 컬렉션은 데이터가 추가/삭제됨에 따라 크기가 자동으로 늘어나고 줄어듭니다.
 * 
 * 2. List의 주요 특징
 *    - 순서(인덱스, 0부터 시작)가 유지됩니다.
 *    - 데이터의 중복 저장을 허용합니다.
 */
public class List_08 {

	public static void main(String[] args) {
		// TODO Auto-generated method stub

		// -------------------------------------------------------------
		// 1. ArrayList 객체 생성 (비제네릭 방식 - 옛날 방식)
		// -------------------------------------------------------------
		// 타입을 지정하지 않고 생성하면 내부적으로 모든 데이터를 최상위 클래스인 'Object'로 저장합니다.
		// 즉, 숫자, 문자열, 불리언 등 어떤 종류의 자료형이든 자유롭게 섞어서 담을 수 있습니다.
		ArrayList list = new ArrayList();

		// -------------------------------------------------------------
		// 2. 요소 추가: add(값)
		// -------------------------------------------------------------
		list.add(1);      // 정수 1 추가 (자바가 자동으로 기본형 int -> 참조형 Integer 객체로 감싸주는 '오토박싱' 수행)
		list.add("글씨");  // 문자열 "글씨" 추가 (String 객체)
		list.add(true);   // 불리언 true 추가 (기본형 boolean -> Boolean 객체로 오토박싱)

		// -------------------------------------------------------------
		// 3. 요소 꺼내기: get(인덱스 번호)
		// -------------------------------------------------------------
		// 인덱스는 배열과 마찬가지로 0번부터 시작합니다.
		System.out.println(list.get(0)); // 0번 위치의 값: 1
		System.out.println(list.get(1)); // 1번 위치의 값: "글씨"
		System.out.println(list.get(2)); // 2번 위치의 값: true

		// -------------------------------------------------------------
		// 4. 리스트의 크기(저장된 데이터 개수): size()
		// -------------------------------------------------------------
		// 일반 배열의 .length 와 동일한 역할을 메서드로 제공합니다.
		System.out.println(list.size()); // 현재 3개의 요소가 저장되어 있으므로 3 출력

		// -------------------------------------------------------------
		// 5. 반복문으로 list의 모든 내용을 출력하기
		// -------------------------------------------------------------
		
		// 방법 (1): 전통적인 for문 (인덱스 i를 이용)
		// 0번 인덱스부터 list.size() - 1 번 인덱스까지 하나씩 순회합니다.
		for (int i = 0; i < list.size(); i++) {
			System.out.println(list.get(i));
		}

		// 방법 (2): 향상된 for문 (for-each문)
		// list 안에 들어있는 요소를 처음부터 끝까지 하나씩 꺼내어 obj 변수에 담아 실행합니다.
		// 비제네릭 리스트이므로 꺼낸 요소는 가장 최상위 부모 타입인 'Object' 타입으로 받습니다.
		for (Object obj : list) {
			System.out.println(obj);
		}
		
		// -------------------------------------------------------------
		// 6. 비제네릭 방식의 단점: 형변환(Casting)의 번거로움
		// -------------------------------------------------------------
		// list.get(0)의 반환 타입은 'Object'입니다.
		// 따라서 기본형 int 변수에 바로 대입할 수 없고, 반드시 (int)로 강제 형변환(Casting)을 해주어야 합니다.
		int a = (int) list.get(0);
		String b = (String) list.get(1); // 1번 인덱스는 문자열이므로 (String) 형변환 필요
		
		// -------------------------------------------------------------
		// 7. 제네릭(Generic, <타입>)의 도입
		// -------------------------------------------------------------
		// 제네릭이란 리스트에 담을 데이터의 타입을 미리 지정해두는 문법입니다.
		// <String>을 지정하면 "이 리스트에는 오직 문자열(String)만 담겠다"는 약속입니다.
		ArrayList<String> list2 = new ArrayList<String>();
		list2.add("문자"); // String 타입이므로 정상 추가됨
		
		// [주의/에러 발생 이유]
		// 아래 코드를 실행하면 컴파일 오류가 납니다.
		// list2는 문자열 전용 리스트로 선언되었기 때문에, 정수(1)를 넣으려고 하면 자바 컴파일러가 입장을 차단합니다.
		// -> 컴파일 시점에 타입을 체크해주어 사전에 오류를 방지할 수 있습니다.
//		list2.add(1); 
		
		// [제네릭의 장점]
		// 요소를 꺼낼 때 번거롭게 (String) 형변환을 하지 않아도,
		// 자바가 이미 String 타입인 것을 알고 있으므로 곧바로 String 변수에 대입할 수 있습니다!
		String c = list2.get(0);
		
		// -------------------------------------------------------------
		// 8. 다형성을 활용한 인터페이스 선언 (실무 표준 권장 방식)
		// -------------------------------------------------------------
		// List는 상위 인터페이스이고, ArrayList는 이를 구현한 하위 클래스입니다.
		// 부모 인터페이스 타입(List)으로 참조 변수를 선언해두면,
		// 나중에 LinkedList 등 다른 List 구현체로 교체할 때도 코드를 유연하게 유지할 수 있습니다.
		List list3 = new ArrayList();
		
		// [주의/예외 발생 이유]
		// 현재 list2에는 0번 인덱스(1개)만 저장되어 있습니다.
		// 존재하지 않는 100번 인덱스를 조회하려고 하면 
		// 실행 도중 'IndexOutOfBoundsException'(인덱스 범위 초과) 예외가 발생하며 프로그램이 비정상 종료됩니다.
//		list2.get(100);

	}

}

package var_02;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * ============================================================================
 * [Map_09.java - 컬렉션 프레임워크: Map(HashMap) & 실무 데이터 구조화]
 * ============================================================================
 * 
 * 1. Map 컬렉션이란?
 *    - 키(Key)와 값(Value)의 쌍(Pair, 엔트리)으로 데이터를 저장하는 자료구조입니다.
 *    - 사물함 열쇠(Key)로 사물함 안의 물건(Value)을 찾는 것과 같습니다.
 * 
 * 2. Map의 핵심 특징
 *    - Key(키) : 중복될 수 없습니다. (유일성 보장, 내부적으로 Set 구조)
 *    - Value(값) : 중복 저장이 가능합니다. (키가 다르면 값은 같아도 상관없음)
 *    - 순서를 보장하지 않습니다. (입력한 순서대로 출력되지 않을 수 있음)
 */
public class Map_09 {

	public static void main(String[] args) {
		// TODO Auto-generated method stub
		
		// -------------------------------------------------------------
		// 1. HashMap 객체 생성 (비제네릭 방식)
		// -------------------------------------------------------------
		// 키와 값 모두 Object 타입으로 처리되므로 어떤 자료형이든 자유롭게 담을 수 있습니다.
		HashMap map = new HashMap();
		
		// -------------------------------------------------------------
		// 2. 데이터 저장: put(키, 값)
		// -------------------------------------------------------------
		map.put("k1" , "v1");   // Key: "k1", Value: "v1" (문자열)
		map.put("k2" , 2);      // Key: "k2", Value: 2 (정수, 오토박싱)
		map.put("k3" , false);  // Key: "k3", Value: false (불리언)
		
		// [Map의 중요한 동작 원리: Key 중복 시 덮어쓰기]
		// 기존에 이미 "k2"라는 Key가 존재하므로, 새로운 데이터가 추가되는 것이 아니라
		// 기존 "k2"의 값(2)이 새로운 값(3)으로 '수정(Update, 덮어쓰기)'됩니다.
		map.put("k2" , 3);
		
		// -------------------------------------------------------------
		// 3. 데이터 조회: get(키)
		// -------------------------------------------------------------
		// Key를 전달하면 그에 연결된 Value를 반환합니다.
		System.out.println(map.get("k1")); // "v1" 출력
		
		// 비제네릭 맵에서 꺼낸 값은 'Object' 타입이므로, 원래 타입으로 쓰려면 형변환이 필요합니다.
		String s = (String)map.get("k1");
		
		// [없는 Key를 조회할 때의 동작]
		// 존재하지 않는 Key("k100")를 get()으로 조회하면 에러(예외)가 나는 것이 아니라 'null'을 돌려줍니다.
		System.out.println(map.get("k100")); // null 출력
		
		// -------------------------------------------------------------
		// 4. 제네릭 Map 선언 (실무 표준 방식)
		// -------------------------------------------------------------
		// Map<String, Integer>: "Key는 반드시 문자열(String), Value는 반드시 정수(Integer)만 넣겠다"는 의미입니다.
		// Map(인터페이스) 타입으로 선언하고 new HashMap()(구현 클래스)으로 생성합니다.
		Map<String, Integer> map2 = new HashMap();
		map2.put("k1", 1);
		
		// [주의/컴파일 에러 이유]
		// Value의 제네릭 타입을 Integer로 지정했기 때문에, 문자열 "1"을 넣으려고 하면 컴파일 에러가 발생합니다.
//		map2.put("k2", "1");
		
		// 제네릭 덕분에 꺼낼 때도 강제 형변환 없이 깔끔하게 Integer 변수로 바로 받을 수 있습니다.
		Integer a = map2.get("k2");
		
		// -------------------------------------------------------------
		// 5. 실무 응용 예제: 멜론 차트 (List 안에 Map 넣기)
		// -------------------------------------------------------------
		// - 멜론 차트 순위 목록(1위, 2위, 3위...) : 순서가 중요하므로 'List'로 관리
		// - 각 곡의 상세 정보(가수명, 제목...) : "제목":"With", "가수":"리센느" 처럼 항목별 매핑이 필요하므로 'Map'으로 관리
		// -> 즉, 엑셀 표의 각 '행(Row)'은 Map이 되고, 표 전체 '목록'은 List가 됩니다!
		List<Map<String, String>> melon = new ArrayList<Map<String, String>>();
		
		// 곡 정보(Map) 하나 생성
		Map<String, String> song = new HashMap<String, String>();
		song.put("제목", "With");
		song.put("가수" , "리센느");
		
		// 리스트에 곡 정보 맵을 추가 (1번 곡으로 등록)
		melon.add(song);
//		System.out.println("melon:" + melon);
		
		// 리스트를 순회하며 멜론 차트에 등록된 모든 곡의 정보를 출력
		for (int i=0; i< melon.size(); i++) {
			// 리스트에서 i번째 곡(Map)을 꺼내옴
			Map map3 = melon.get(i);
			System.out.println("map3: " + map3);
			// 꺼내온 Map에서 Key("제목", "가수")를 이용해 실제 값을 조회
			System.out.println("제목: " + map3.get("제목"));
			System.out.println("가수: " + map3.get("가수"));
		}
		
		// -------------------------------------------------------------
		// 6. final 키워드 (상수 만들기)
		// -------------------------------------------------------------
		// 일반 변수: 값을 언제든지 다른 값으로 변경(재대입)할 수 있습니다.
		int x = 10;
		x = 20; // 10 -> 20으로 값 변경 성공
		
		// final 변수: '최종적인 값'이라는 뜻으로, 한 번 초기화되면 값을 절대 바꿀 수 없는 '상수(Constant)'가 됩니다.
		final int y = 20;
		
		// [주의/컴파일 에러 이유]
		// y는 final로 선언되었으므로 값을 다시 대입하려고 하면 컴파일 에러("cannot assign a value to final variable")가 발생합니다.
//		y = 200;
		
		
	}

}

package class_01;

/**
 * ============================================================================
 * [Car_02.java - 객체지향 프로그래밍: 클래스 설계, 필드, 메서드, 오버로딩, 생성자]
 * ============================================================================
 * 
 * 1. 클래스(Class)란?
 *    - 객체(인스턴스)를 만들기 위한 '설계도' 또는 '틀'입니다.
 *    - 클래스는 속성(상태)을 나타내는 '필드(Field)'와 동작(기능)을 나타내는 '메서드(Method)'로 구성됩니다.
 */
public class Car_02 {
	
	// ========================================================================
	// [1] 필드 (Field / 멤버 변수 / 전역 변수)
	// ========================================================================
	// - 객체가 가지는 고유한 속성(데이터)을 저장하는 변수입니다.
	// - 필드는 선언만 하거나, 선언과 동시에 초기값을 대입할 수 있습니다.
	
	String Company = "쉐보레"; // 선언과 동시에 명시적 초기화
//	String model = "임팔라";
	String model;              // 선언만 한 상태 (참조 타입이므로 기본값 null로 자동 초기화)
	int maxSpeed = 240;        // 최고 속도 240으로 초기화
	int year;                  // 기본값 0으로 자동 초기화
	
	// [필드의 기본 초기값 규칙]
	// 로컬 변수(메서드 안의 변수)는 초기화하지 않으면 사용할 수 없지만,
	// 클래스의 '필드'는 초기화하지 않아도 객체 생성(new) 시 자바가 기본값으로 자동 초기화해 줍니다.
	// - 정수형(byte, short, int, long) : 0
	// - 실수형(float, double) : 0.0
	// - 논리형(boolean) : false
	// - 참조형(String, 배열, 객체 등) : null
	int speed; // 초기화하지 않았으므로 0으로 자동 초기화됨
	
	// [주의: 필드 영역의 문법 규칙]
	// 클래스 블록 바로 아래(필드 영역)에서는 오직 '선언'과 '초기화 대입'만 가능합니다.
	// 연산, 메서드 호출, 조건문, 단순 대입문 같은 '실행(동작) 코드'는 반드시 메서드나 생성자 블록 내부에서만 작성해야 합니다.
//	speed = 10; // 컴파일 에러: 필드 영역에서는 단독 실행문 작성 불가!
	
	boolean power; // 기본값 false로 자동 초기화됨 (시동 꺼짐 상태)
	
	// ========================================================================
	// [2] 메서드 (Method / 기능)
	// ========================================================================
	// - 객체가 수행할 동작이나 행위를 정의하는 블록입니다.
	// - 문법: [리턴타입] [메서드명](<[전달인자/매개변수 목록]>) { 실행할 코드 }
	
	// [시동 켜기 메서드]
	// void: 실행 후 호출한 곳으로 돌려줄 값(반환값)이 없음을 의미합니다.
	void powerOn() {
		power = true; // 시동 상태를 true(켜짐)로 변경
	}
	
	// [시동 끄기 메서드]
	void powerOff() {
		power = false; // 시동 상태를 false(꺼짐)로 변경
	}
	
	// [속도 설정 세터(Setter) 메서드]
	// 전달받은 매개변수 s 값을 speed 필드에 반영합니다.
	// 단, 안전을 위해 최고 속도(maxSpeed)를 넘어서면 최고 속도로 제한합니다.
	void setSpeed(int s) {
		System.out.println("s:" + s);
		
		if (s >= maxSpeed) {
			s = maxSpeed; // 최고 속도로 고정
			System.out.println("그만 올려!!");
		}
		speed = s;
		return; // void 메서드에서 return;은 메서드를 즉시 종료하고 빠져나간다는 의미 (생략 가능)
	}
	
	/*
	 * [속도 조회 게터(Getter) 메서드]
	 * 현재 필드의 speed 값을 돌려줍니다.
	 * @param 없음 
	 * @return int (현재 속도 값)
	 */
	int getSpeed() {
		return speed; // int 타입의 speed 값을 호출한 곳으로 반환
	}
	
	// [매개변수를 받는 getSpeed 오버로딩]
	int getSpeed(int s) {
		System.out.println();
		System.out.println(s);
		return speed;
	}
	
	// ========================================================================

 *      모든 객체들이 하나의 값을 '공동으로 공유'할 때 사용합니다.
 *    - 호출 방법: 클래스명.정적멤버명 (예: Static_03.PI)
 */
public class Static_03 {
	// ========================================================================
	// [1] static 변수 (정적 변수 / 클래스 변수)
	// ========================================================================
	// new로 객체를 생성하지 않고도 클래스 이름(Static_03.PI)으로 바로 접근할 수 있습니다.
	// 이 클래스로부터 만들어지는 모든 객체가 같은 PI 값을 함께 공유합니다.
	static double PI = 3.14;
	
	// [인스턴스 변수 (일반 필드)]
	// static이 붙지 않은 변수는 반드시 new 연산자로 객체를 생성해야만 힙 메모리에 생깁니다.
	// 객체마다 독립된 name 공간을 갖습니다.
	String name;
	
	// ========================================================================
	// [2] 상수 선언 : static final
	// ========================================================================
	// - static : 객체마다 따로 만들 필요 없이 클래스 차원에서 단 하나만 존재 (메모리 절약)
	// - final  : 한 번 정해진 값을 절대로 변경할 수 없음 (불변성 보장)
	// -> 즉, 프로그램 전체에서 공유하며 절대 바뀌면 안 되는 '공용 고정값'을 만들 때 사용합니다.

	static double getPI()
