``` java
		 * ====================================================================
		 * - 원본 b = {7, 12, 8, 5} -> 뒤집은 결과 c = {5, 8, 12, 7}
		 * - 공식 유도:
		 *   c[0] = b[3] -> 인덱스 합이 항상 (length - 1)
		 *   따라서 c[i] = b[b.length - 1 - i]

		int [] b = {7,12,8,5};

		c[0] = b[3]; // length - (0+1) // length - 1 - 0
		c[1] = b[2]; // length - (1+2) // length - 1 - 1
		c[2] = b[1]; // length - (2+1)

		 *   기존 데이터를 복사해 넣고 마지막 칸에 새 값을 넣어야 해.
		 */
		b = new int[] {7, 12, 8, 5};
		int new_value = 3;
		
		// 크기가 5(4 + 1)인 새로운 배열 생성
		int [] d = new int[b.length + 1];
		// 기존 b 배열의 값들을 그대로 복사
		for (int i=0; i<b.length; i++) {
			d[i] = b[i];
		}
		// 새로 늘어난 마지막 칸(인덱스 4)에 새 값 저장
		d[d.length - 1] = new_value;
		
		for (int i = 0; i < d.length; i++) {
            System.out.println("d[" + i + "] = " + d[i]);
        }
		
		
		/* ====================================================================
		 * 8. 배열 실습 (3) : 배열 맨 앞(0번 인덱스)에 3을 추가하기
		 * ====================================================================
		 * - 0번 자리에 새 값을 넣고, 기존 값들은 전부 한 칸씩 뒤로 밀어야 해 (i+1).
		 */
		int[] e = new int[b.length+1];
		e[0] = 3; // 맨 앞 칸에 3 삽입
		
		// 기존 값들을 1번 인덱스부터 차례대로 복사
		for (int i=0; i<b.length; i++) {
			e[i+1] = b[i];
		}
		
		
		/* ====================================================================
		 * 9. 배열 실습 (4) : 배열 내 조건 만족 개수 세기 (홀수의 개수)
		 * ====================================================================
		 * - % 2 == 1 은 홀수를 판별하는 공식이야.
		 */
		int count = 0;
		for (int i=0; i<e.length; i++) {
			if (e[i] % 2 == 1) {
				count++; // 홀수일 때마다 카운트 1 증가
			}
		}
		
		
		/* ====================================================================
		 * 10. 배열 실습 (5) : 배열의 특정 위치(인덱스 2번)에 값 삽입하기
		 * ====================================================================
		 * - [참고 노트]
		 *   원래 새 값을 삽입하면 총 길이가 1 늘어나야 하므로 new int[b.length + 1]이 정석이야.
		 *   (아래 코드는 크기가 b.length 그대로라서 마지막 요소가 잘리게 됨을 염두에 둘 것!)
		 * - 로직 흐름:
		 *   1) i < index  : 삽입 위치 전이므로 그대로 복사 f[i] = b[i]
		 *   2) i == index : 지정한 위치이므로 새 값 삽입 f[i] = value
		 *   3) i > index  : 삽입 위치 뒤이므로 앞 칸의 값을 당겨옴 f[i] = b[i-1]
		 */
		int index = 2;
		int value = 3;
		int [] f = new int[b.length];
		for (int i=0; i<f.length; i++) {
			if (i == index) {
				f[i] = value;
			
			}else if(i < index) {
				f[i] = b[i];
			}else if(i > index) {
				f[i] = b[i-1];
			}
		}
		for(int i=0; i<f.length; i++) {
			System.out.println(f[i]);
		}
   		
		/* ====================================================================
		 * 11. 주말 숙제 힌트 가이드
		 * ====================================================================
		 * [문제] 10자리 임시 비밀번호 만들기 (대문자 1개 이상, 소문자 1개 이상, 숫자 1개 이상 포함)
		 * [힌트]
		 * - 컴퓨터는 문자를 '아스키코드(숫자)'로 다뤄.
		 *   '0'~'9' : 48 ~ 57
		 *   'A'~'Z' : 65 ~ 90
		 *   'a'~'z' : 97 ~ 122
		 * - Math.random()으로 위 아스키코드 범위의 정수를 뽑은 뒤 (char)로 형변환하면
		 *   임의의 알파벳/숫자 문자를 얻을 수 있어.
		 */
	}

}
```

``` java
package var_02;

public class If_04 {

	public static void main(String[] args) {
		
		/* ====================================================================
		 * 1. if - else if - else 기본 흐름
		 * ====================================================================
		 * - if (조건1)      : 조건1이 참이면 실행
		 * - else if (조건2) : 앞의 조건이 거짓이고 조건2가 참이면 실행
		 * - else            : 위의 모든 조건이 거짓일 때 실행
		 * - 위에서 아래로 순차 검사하며, 한 번이라도 조건을 만족하면 아래는 건너뜀!
		 */
		if(3 < 2) {
			System.out.println("3 < 2"); // 거짓이므로 통과
		}else if (3 != 3) {
			System.out.println("3 != 2"); // 거짓(3은 3과 같음)이므로 통과
		}else {
			System.out.println("else");   // 최종 실행됨
		}
		
		// 단순 if문 (조건이 참일 때만 실행)
		if(3 > 2) {
			System.out.println("3 > 2");
		}
		
		// 실행할 문장이 1줄일 때는 중괄호 {}를 생략할 수 있어.
		// (하지만 가독성과 실수를 방지하기 위해 평소엔 중괄호를 쓰는 게 좋아)
		if (3 > 2) System.out.println("3 > 2");
		else 	   System.out.println("거짓");
		
		
		/* ====================================================================
		 * 2. 기습문제 : 짝수 / 홀수 판별하기
		 * ====================================================================
		 * - % (나머지 연산자) : 어떤 수를 2로 나눈 나머지가 1이면 홀수, 0이면 짝수
		 */
		int a = 3;
		if (a % 2 == 1) {
			System.out.println("홀");
		} else {
			System.out.println("짝");
		}
	
		
		/* ====================================================================
		 * 3. Math.random()을 이용한 난수(랜덤 숫자) 생성 단계별 원리
		 * ====================================================================
		 * - Math.random()은 0.0 이상 1.0 미만(0.0 <= x < 1.0)의 double 실수를 반환해.
		 */
		double rand = Math.random();
		int min = 3;
		int max = 7;
		
		// 0.0 * 5 <= rand < 1.0 * 5  ==>  0.0 <= rand < 5.0
		rand = Math.random() * 5;
		System.out.println(rand);
		
		// [1단계] 0 ~ 7(max)까지의 정수 나오게 만들기
		// 1) 0.0 <= Math.random() < 1.0 에 (max + 1, 즉 8)을 곱하면: 0.0 <= x < 8.0
		// 2) (int)로 강제 형변환하여 소수점을 버리면: 0 ~ 7 정수가 됨
		rand = Math.random() * (max+1);
		rand = (int)(Math.random() * (max+1));
		
		// [2단계] 3(min) ~ 10까지 나오게 만들기
		// 0 ~ 7 범위의 정수에 최솟값 min(3)을 더하면 3 ~ 10이 됨
		rand = (int) (Math.random() * (max+1)) + min;
		
		// [3단계] 3(min) ~ 7(max)까지 나오게 만들기 (공식 유도)
		// * 핵심 공식 : (int)(Math.random() * (max - min + 1)) + min
		//   - 개수(경우의 수) = (max - min + 1) = 7 - 3 + 1 = 5개 (3, 4, 5, 6, 7)
		//   - (0~4까지 나온 정수) + 시작값(3) = 3 ~ 7
		// * 주의: 아래 코드의 ((min-max)+1)은 오타로 인해 음수가 나오므로 ((max-min)+1)이 올바른 공식이야!
		rand = (int) (Math.random() * ((min-max)+1)) + min;
		
		System.out.println(rand);
		
		// [번외 방법] 큰 난수를 생성한 뒤 나머지 연산자(%)로 범위 맞추기
		// 10000 이하의 큰 수를 경우의 수(5)로 나눈 나머지(0~4) + 시작값(3) -> 3 ~ 7
		rand = (int)(Math.random() * 10000) % ((max-min)+1) + min;
		
		
		/* ====================================================================
		 * 4. 0으로 나누기 오류와 단락 평가(Short-Circuit Evaluation)
		 * ====================================================================
		 */
		int num = 3;
		int num2 = 0;
		
		// [위험한 코드 예시]
		// 어떤 숫자든 0으로 나누면(/) 컴퓨터는 계산할 수 없어서 
		// 프로그램이 즉시 멈추고 런타임 에러(ArithmeticException: / by zero)를 발생시켜!
		if (num / num2 == 0) {
			System.out.println("참");
		}else {
			System.out.println("거짓");
		}
		
		// [안전한 코드 : 단락 평가(Short-circuit evaluation) 활용]
		// AND(&&) 연산자는 앞의 조건이 false이면, 뒤의 조건은 아예 쳐다보지도 않고 전체를 false로 결론내!
		// 따라서 num2 != 0 이 false(0이라는 뜻)이므로, 뒤쪽의 위험한 식(num / num2)을 아예 실행하지 않아
		// 0으로 나누는 에러를 안전하게 방지할 수 있어.
		if (num2 != 0 && num / num2 == 0) {
			System.out.println("참");
		}else {
			System.out.println("거짓");
		}
		
	}

}
```

``` java
package var_02;

import java.util.Scanner;

public class Switch_05 {
	public static void main(String[] args) {
		
		/* ====================================================================
		 * 1. Scanner를 통한 사용자 키보드 입력 받기
		 * ====================================================================
		 * - Scanner scan = new Scanner(System.in);
		 *   콘솔 창에서 사용자가 키보드로 타이핑한 값을 읽어오는 객체
		 * - scan.nextInt() : 사용자가 입력한 정수(int)를 읽어서 변수 a에 저장
		 */
		System.out.println("숫자를 입력하세요:");
		Scanner scan = new Scanner(System.in);
		int a = scan.nextInt();
		System.out.println("a:" + a);
		
		
		/* ====================================================================
		 * 2. switch - case 문의 기본 구조와 break의 중요성
		 * ====================================================================
		 * - switch(변수) : 변수의 값과 일치하는 case 라벨로 바로 점프해서 실행해.
		 * - break; 의 역할:
		 *   switch 문 블록을 즉시 빠져나가라는 탈출 명령어!
		 *   ★ 만약 break가 없으면? 일치한 case 이후의 다른 case문들까지 조건 검사 없이
		 *     break를 만날 때까지 연달아 실행되어 버려 (이를 Fall-through 현상이라고 해).
		 *     (아래 코드에서 a가 1이면 case 1 출력 후 break가 없어서 case 2까지 출력됨!)
		 * - default : 일치하는 case가 하나도 없을 때 실행되는 기본 구간 (if문의 else와 같음).
		 */
		switch(a) {
			case 1 :
				System.out.println("a는 1이다");
				// break가 없으므로 아래 case 2로 흘러내려감!
			case 2 : 
				System.out.println("a는 2이다");
				break; // 여기서 비로소 switch문을 탈출
			default : 
				System.out.println("그 외");
		}
		
		
		/* ====================================================================
		 * 3. switch 조건식에 사용 가능한 자료형 vs 불가능한 자료형
		 * ====================================================================
		 * [사용 가능 O]
		 * - 정수형 : byte, short, char, int
		 * - 문자열 : String (자바 7부터 지원)
		 * - 기타   : Enum(열거형)
		 * 
		 * [사용 불가능 X]
		 * - 실수형 : float, double (부동소수점 오차 때문에 정확한 값 비교 불가)
		 * - 논리형 : boolean (참/거짓은 if문으로만 처리)
		 * - 큰정수 : long (switch의 점프 테이블 구조상 지원 안 함)
		 */
		// 사용 가능 : byte, char, short, int, String
		// 사용 불가능 : boolean, long, float, double
		
		
		/* ====================================================================
		 * 4. 여러 case를 묶어서 처리하기 (Fall-through의 유용한 활용)
		 * ====================================================================
		 * - 12, 1, 2월은 모두 겨울이므로 case들을 연달아 나열하고
		 *   마지막에만 출력문과 break를 배치하여 코드를 깔끔하게 압축할 수 있어.
		 */
		switch(a){
			case 12:
			case 1:
			case 2:
				System.out.println("겨울");
				break;
			case 3:
			case 4:
			case 5:
				System.out.println("봄");
				break;
		}
	}
}
```

``` java
package var_02;

public class For_06 {

	public static void main(String[] args) {
		
		/* ====================================================================
		 * 1. for 반복문의 기본 구조와 실행 순서
		 * ====================================================================
		 * for (초기식; 조건식; 증감식) { 실행문; }
		 * 
		 * [실행 순서]
		 * 1) 초기식 실행 (int i = 0) -> 딱 1번만 실행
		 * 2) 조건식 검사 (i < 10)    -> 참이면 본문 실행, 거짓이면 반복문 종료
		 * 3) 본문 실행 (System.out.println(i))
		 * 4) 증감식 실행 (i++)        -> i가 1 증가한 뒤 다시 2번 조건식으로 이동
		 */
		// 무한 루프의 형태 (조건식을 비워두면 항상 참으로 간주됨)
//		for (; ;) {
//			
//		}
		
		// 0부터 9까지 10회 반복 출력
		for (int i=0; i<10; i++) {
			System.out.println(i);
		}
		
		
		/* ====================================================================
		 * 2. 0부터 10까지 모두 더하기 (누적 합산 알고리즘)
		 * ====================================================================
		 * - 합계를 담을 변수(result)를 반복문 '바깥'에서 0으로 초기화해 두는 것이 핵심!
		 * - result += i  ==>  result = result + i 와 동일한 의미
		 */
		// 0부터 10까지 모두 더하시오
		int result = 0;
		for (int i=0; i<11; i++) { // i가 0부터 10까지 총 11회 회전
			result += i;		
		}
		System.out.println(result); // 0 + 1 + 2 + ... + 10 = 55
		
		
		/* ====================================================================
		 * 3. 문제 1 : 주사위 2개를 굴려서 나올 수 있는 모든 조합 출력 (이중 for문)
		 * ====================================================================
		 * - 바깥쪽 for문(i)이 1번 돌 때, 안쪽 for문(j)은 전체를 다 돎!
		 * - [참고 주의사항]
		 *   원래 주사위 눈은 1~6이므로 조건식이 i<=6, j<=6 이어야 해.
		 *   (아래 코드는 i<6, j<6 이라서 1~5까지만 출력됨에 유의할 것!)
		 */
		// 문제1 
		// 주사위 2개를 굴려서 
		// 나올 수 있는 모든 조합 출력
		for (int i=1; i<6; i++) {
			for (int j=1; j<6; j++) {
				System.out.println("나올 수 있는 모든 출력 :" + i + " " + j);
			}
		}
		
		
		/* ====================================================================
		 * 4. 문제 2 : 주사위 2개를 굴려 합계(k) 별로 조합 출력
		 * ====================================================================
		 * - 두 주사위 합의 최솟값은 1+1=2, 최댓값은 6+6=12.
		 * - 가장 바깥 for문(k)으로 합계 2부터 12까지 기준을 잡고,
		 *   안쪽의 주사위 조합(i, j) 중 합이 k와 같은 것만 골라 출력함.
		 * - printf("[%d, %d]", i, j) : 서식에 맞춰 깔끔하게 가로로 출력 (%d는 정수)
		 */
		// 문제2
		// 주사위 2개를 굴려서
		// 나올 수 있는 모든 조합을 
		// 합 별로 출력
		for (int k=2; k<=12; k++) {
			System.out.print(k + ":");
			for(int i=1; i<=6; i++) {
				for (int j=1; j<=6; j++) {
					if (i+j == k) {
						System.out.printf("[%d, %d]" , i, j);
					}
				}
			}
			System.out.println(); // 줄바꿈
		}
		
		
		/* ====================================================================
		 * 5. 문제 3 : 합계별 주사위 조합 출력 (단, 중복 제거)
		 * ====================================================================
		 * - 주사위 눈 [1, 2]와 [2, 1]은 순서만 다르고 구성 숫자가 같으므로 중복이야.
		 * - 중복 제거 핵심 조건: `i <= j`
		 *   첫 번째 주사위(i)가 두 번째 주사위(j)보다 작거나 같을 때만 허용하면,
		 *   [1, 2]는 통과하고 [2, 1]은 걸러져서 깔끔하게 중복이 사라져!
		 */
		// 문제3
		// 주사위 2개를 굴려서
		// 나올 수 있는 모든 조합을 
		// 합 별로 출력
		// 단, 중복 없이
		for (int k=2; k<=12; k++) {
			System.out.print(k + ":");
			for(int i=1; i<=6; i++) {
				for (int j=1; j<=6; j++) {
					if (i+j == k && i <= j ) { // i <= j 조건으로 뒤집힌 중복 제거
						System.out.printf("[%d, %d]" , i, j);
					}
				}
			}
			System.out.println(); // 줄바꿈
		}
	}

}
```

## While_07.java 
``` java
package var_02;

public class While_07 {

	public static void main(String[] args) {
		
		/* ====================================================================
		 * 1. while 반복문의 기본 구조
		 * ====================================================================
		 * - while(조건식) : 조건이 참(true)인 동안 계속해서 중괄호 내부를 반복 실행.
		 * - ★ 주의 : 반복문 안에 i++ 같은 '증감식'이 빠지면 i의 값이 변하지 않아
		 *   조건이 영원히 참이 되어 컴퓨터가 멈추지 않는 '무한 루프(Infinite Loop)'에 빠져!
		 */
		int i = 3;
		while(i < 10) {
			System.out.println(i);
			i++; // i를 1씩 증가 (3 -> 4 -> ... -> 10이 되면 탈출)
		}
		
		
		/* ====================================================================
		 * 2. do - while 반복문 (최소 1회 실행 보장)
		 * ====================================================================
		 * - 일반 while문은 시작하자마자 조건을 먼저 검사하지만,
		 *   do - while문은 조건을 '맨 마지막'에 검사해.
		 * - 따라서 조건이 처음부터 거짓(false)이더라도 본문을 '무조건 최소 1번'은 실행해!
		 *   (위에서 i가 이미 10이 된 상태지만, do 블록 안의 10이 1번 출력되고 i는 11이 됨)
		 */
		do {
			System.out.println(i);
			i++;
		}while(i < 10);
		
		System.out.println("----------------------");
		
		
		/* ====================================================================
		 * 3. continue 키워드를 활용한 짝수 출력
		 * ====================================================================
		 * - break    : 반복문을 그 즉시 완전히 중단하고 '탈출'
		 * - continue : 이번 턴의 남은 코드를 건너뛰고, 곧바로 '다음 반복 회차'로 넘어감!
		 * - 아래 코드 흐름:
		 *   i가 홀수(i % 2 != 0)이면 continue를 만나 밑의 println을 건너뛰고
		 *   다시 while 조건문으로 올라가므로 결과적으로 '짝수'만 화면에 출력돼.
		 */
		i = 0;
		while(i < 10) {
			i++;
			if(i % 2 != 0) { // 홀수 검증 조건
				continue;    // 홀수면 아래의 println을 건너뜀
			}
			System.out.println(i); // 짝수(2, 4, 6, 8, 10)만 출력됨
			
		}

	}

}
```
