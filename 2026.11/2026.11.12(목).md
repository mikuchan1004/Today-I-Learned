package com.example.demo;

import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;

/**
 * [스프링 부트 애플리케이션의 시작 클래스]
 * 
 * @SpringBootApplication: 스프링 부트의 가장 핵심적인 어노테이션으로 다음 3가지 기능을 한 번에 수행합니다.
 *   1. @SpringBootConfiguration : 스프링 부트 설정 클래스임을 선언
 *   2. @EnableAutoConfiguration : 복잡한 설정을 자동으로 처리 (웹 서버, DB 드라이버 등 자동 구성)
 *   3. @ComponentScan           : 현재 패키지(com.example.demo)와 하위 패키지에서 @Controller, @Service, @Repository 등의 빈(Bean)을 찾아 스프링 컨테이너에 등록
 */
@SpringBootApplication
public class BootHello01Application {

	/**
	 * 자바 애플리케이션의 시작점(main 메서드)
	 * - SpringApplication.run()을 실행하면 내장 톰캣(Tomcat) 서버가 자동으로 시작되며 웹 애플리케이션이 구동됩니다.
	 */
	public static void main(String[] args) {
		System.out.println("hello world");
		// 현재 클래스를 전달하여 스프링 부트 환경을 초기화하고 실행
		SpringApplication.run(BootHello01Application.class, args);
	}

}

package com.example.demo.dto;

import java.sql.Date;
import java.util.List;

import lombok.Data;

/**
 * [사원 정보 DTO (Data Transfer Object)]
 * - 계층 간(Controller, Service, DAO, MyBatis 매퍼, JSP 화면) 데이터를 전달하는 바구니 객체
 * - 데이터베이스 EMP 테이블의 컬럼들과 1:1 매핑되는 변수들과
 *   검색(동적 쿼리), 페이징 처리를 위해 필요한 추가 변수들을 함께 담고 있습니다.
 */
public class EmpDTO {

	// ==========================================
	// 1. EMP 테이블의 컬럼과 대응되는 필드들
	// ==========================================
	private int empno;                    // 사원 번호 (기본키)
	private String ename;                 // 사원 이름
	private String job;                   // 직무/담당 업무
	private int sal = Integer.MIN_VALUE;  // 급여 (동적 쿼리에서 0 이상의 값이 들어왔는지 구분하기 위해 최소값으로 초기화)
	private Integer mgr;                  // 직속 상사의 사원번호 (null 허용이므로 Wrapper 클래스인 Integer 사용)
	private Date hiredate;                // 입사일자
	private Integer comm;                 // 보너스/커미션 (null 허용)
	private int deptno;                   // 부서 번호
	
	// ==========================================
	// 2. 페이징 처리를 위한 필드들 (게시판/목록 분할용)
	// ==========================================
	private int size = 10;                // 한 페이지당 보여줄 게시글/사원 수
	private int page = 1;                 // 현재 보고 있는 페이지 번호
	private int start = 0;                // DB 조회 시작 row 번호 (LIMIT/OFFSET 등에서 사용)
	private int end;                      // 끝 row 번호
	
	// ==========================================
	// 3. 동적 검색(MyBatis dynamic query)용 필드들
	// ==========================================
	private String type;                  // 검색 조건 종류 (예: 'ename', 'sal', 'job')
	private String keyword;               // 사용자가 입력한 검색어
	
	// ==========================================
	// 4. 다중 선택 검색용 (MyBatis <foreach> 실습용)
	// ==========================================
	private List empnos;                  // 사원번호 목록 (예: [7369, 7499, 7521] -> IN 절 조건 생성)


	public int getEmpno() {
		return empno;
	}

	public void setEmpno(int empno) {
		this.empno = empno;
	}

	public String getEname() {

	public void setEname(String ename) {
		this.ename = ename;
	}

	public String getJob() {
		return job;
	}

	public void setJob(String job) {

	}

	public int getSal() {

package com.example.demo.controller;

import java.util.List;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.ModelAttribute;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestMethod;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.ResponseBody;

import com.example.demo.dto.EmpDTO;
import com.example.demo.service.EmpService;

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpSession;

/**
 * [사원 관리 컨트롤러 (Controller)]
 * - 클라이언트(사용자 웹 브라우저)의 요청(URL)을 가장 먼저 받아 분석하고,
 *   해당 작업을 비즈니스 로직(Service)에 위임한 뒤
 *   결과 데이터를 Model에 담아 뷰(JSP) 화면으로 전달하는 중계자 역할을 합니다.
 */
@Controller // 스프링이 이 클래스를 웹 요청을 처리하는 컨트롤러 빈(Bean)으로 인식하도록 등록
public class EmpController {

	// 로그 출력을 위한 SLF4J 로거 객체 생성
	private static final Logger logger = LoggerFactory.getLogger(EmpController.class);
	
	/**
	 * @Autowired (의존성 주입 - DI, Dependency Injection)
	 * - 개발자가 직접 new EmpServiceImpl()을 호출하지 않아도,
	 *   스프링 컨테이너가 EmpService 인터페이스를 구현한 @Service 빈 객체를 찾아 자동으로 주입해 줍니다.
	 */
	@Autowired
	EmpService empService;
	
	/**
	 * [기본 루트 경로 요청 처리] - http://localhost:8080/
	 * - @ResponseBody: JSP 화면을 거치지 않고, 반환된 문자열("hello")을 브라우저에 그대로 순수 텍스트(HTTP Body)로 응답합니다.
	 */
	@ResponseBody
	@RequestMapping(value="/", method=RequestMethod.GET)
	public String hello() {
		return "hello";
	}
	
	/**
	 * [전체 사원 목록 조회] - GET http://localhost:8080/list
	 * 
	 * @param model : 컨트롤러에서 조회한 데이터를 JSP 뷰 페이지로 전달하기 위해 스프링이 제공하는 데이터 전달 상자
	 * @return "result" -> application.properties에 설정된 prefix(/WEB-INF/jsp/) + "result" + suffix(.jsp)로 결합되어
	 *                    /WEB-INF/jsp/result.jsp 페이지가 렌더링됩니다.
	 */
	@GetMapping("/list")
	public String list(Model model) {
		System.out.println("/list 실행");
		
		// 1. 서비스 계층을 호출하여 전체 사원 목록(List<EmpDTO>)을 가져옵니다.
		List<EmpDTO> list = empService.getEmpList();
		System.out.println("/list: list: "+ list);
		
		// 2. JSP 화면에서 ${list}라는 이름으로 접근할 수 있도록 Model에 담습니다.
		model.addAttribute("list", list);
		
		// 3. result.jsp 화면으로 포워딩(이동)
		return "result";
	}

	/**
	 * [사원 단건 상세 조회] - http://localhost:8080/getEmpno?empno=7788
	 * 
	 * @param model : 조회된 EmpDTO 단건 객체를 JSP에 넘겨주기 위한 모델 객체
	 * @param empno : @RequestParam을 통해 URL 쿼리스트링(?empno=7788)의 값을 정수형 int로 자동 형변환하여 받음
	 *                - required=false: 파라미터가 필수가 아님을 명시 (생략 시 기본 0 등으로 처리 가능)
	 * @return "result" -> /WEB-INF/jsp/result.jsp
	 */
	@RequestMapping("/getEmpno")
	public String getEmpno(Model model,
		@RequestParam(value="empno", required=false)
		int empno
	) {
		System.out.println("/getEmpno 실행");
		
		System.out.println("empno: "+ empno);
		// 1. 특정 사원 번호에 해당하는 사원 1명의 정보를 서비스에서 조회
		EmpDTO empDTO = empService.getEmpno(empno);

		// 2. JSP 화면에서 ${empDTO}라는 이름으로 사원 정보를 꺼낼 수 있도록 담기
		model.addAttribute("empDTO", empDTO);
		
		// 3. result.jsp 화면으로 포워딩 (result.jsp에서 list가 없으면 단건 empDTO를 출력하도록 분기되어 있음)
		return "result";
	}

}

package com.example.demo.service;

import java.util.List;
import java.util.Map;

import com.example.demo.dto.EmpDTO;

/**
 * [사원 비즈니스 로직 인터페이스 (Service Interface)]
 * 
 * [인터페이스를 따로 두는 이유]
 * - 컨트롤러와 실제 구현체(ServiceImpl) 사이에 규격(명세)을 두어 느슨한 결합(Loose Coupling)을 만듭니다.
 * - 나중에 비즈니스 로직의 구현 방식이 바뀌거나 트랜잭션, AOP, 테스트 Mock 객체를 적용할 때
 *   컨트롤러 코드를 고칠 필요 없이 유연하게 교체할 수 있습니다.
 */
public interface EmpService {

	/**
	 * 전체 사원 목록을 조회하는 비즈니스 메서드
	 */
	List<EmpDTO> getEmpList();
	
	/**
	 * 특정 사원번호(empno)에 해당하는 사원 1명의 정보를 조회하는 비즈니스 메서드
	 */
	EmpDTO getEmpno(int empno);
}

package com.example.demo.service;

import java.util.List;
import java.util.Map;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import com.example.demo.dao.EmpDAO;
import com.example.demo.dto.EmpDTO;

/**
 * [사원 비즈니스 로직 구현체 (Service Implementation)]
 * 
 * - @Service: 스프링이 이 클래스를 비즈니스 로직을 수행하는 서비스 빈(Bean)으로 관리하도록 등록합니다.
 * - 컨트롤러로부터 전달받은 요청을 처리하며, 필요에 따라 가공, 계산, 트랜잭션(@Transactional) 처리 등을 수행하고
 *   데이터 접근이 필요하면 DAO(Data Access Object)를 호출합니다.
 */
@Service
public class EmpServiceImpl implements EmpService {

	/**
	 * DAO 객체 주입받기
	 * - 데이터베이스와 직접 통신하는 EmpDAO 구현체(EmpDAOImpl)를 스프링이 자동으로 주입해 줍니다.
	 */
	@Autowired
	EmpDAO empDAO;
	
	/**
	 * 전체 사원 목록 조회 처리
	 */
	@Override
	public List<EmpDTO> getEmpList() {
		// DAO의 selectAllEmp()를 호출하여 DB에서 사원 전체 데이터를 조회해 반환
		List<EmpDTO> result = empDAO.selectAllEmp();
		return result;
	}

	/**
	 * 사원번호 기준 단건 상세 조회 처리
	 */
	@Override
	public EmpDTO getEmpno(int empno) {
		// DAO의 selectEmpno()를 호출하여 특정 사원의 정보를 DB에서 가져옴
		EmpDTO empDTO = empDAO.selectEmpno(empno);
		return empDTO;
	}

}

package com.example.demo.dao;

import java.util.List;

import com.example.demo.dto.EmpDTO;

/**
 * [사원 데이터 접근 객체 인터페이스 (DAO, Data Access Object)]

 * - 데이터베이스(DB)와의 CRUD(생성, 조회, 수정, 삭제) 작업을 전담하는 계층의 규격입니다.
 * - Service 계층은 오직 이 DAO 인터페이스의 메서드만 호출하며,
 *   실제 어떤 DB 기술(MyBatis, JPA 등)을 사용하는지는 신경 쓰지 않아도 되는 구조를 만듭니다.

	/**
	 * 전체 사원 목록을 DB에서 조회하는 DAO 메서드

	List<EmpDTO> selectAllEmp();

package com.example.demo.dao;

import java.util.List;
import java.util.Map;

import org.apache.ibatis.session.SqlSession;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Repository;

import com.example.demo.dto.EmpDTO;

/**
 * [사원 데이터 접근 객체 구현체 (DAO Implementation)]
 * 
 * - @Repository: 스프링이 이 클래스를 데이터베이스 접근 계층(DAO) 빈(Bean)으로 등록하며,
 *   DB 관련 예외를 스프링의 DataAccessException으로 일관되게 변환해 주는 기능도 제공합니다.
 */
@Repository
public class EmpDAOImpl implements EmpDAO {

	/**
	 * MyBatis 핵심 세션 객체 주입
	 * - SqlSession: SQL 실행, 트랜잭션 관리, 매퍼 XML 파일 호출을 담당하는 MyBatis의 핵심 인스턴스
	 * - 스프링 부트 mybatis-spring 연동 라이브러리가 스레드에 안전한(SqlSessionTemplate) 객체로 자동 주입해 줍니다.
	 */
	@Autowired
	SqlSession sqlSession;
	
	/**
	 * [전체 사원 조회]
	 * - sqlSession.selectList(매퍼_식별자)
	 *   : 매퍼 XML의 namespace가 "mapper.emp"이고 id가 "selectEmp"인 <select> 태그를 찾아 실행합니다.
	 *   : 조회 결과가 여러 건(List)일 때 사용하며, 각 행(Row)은 EmpDTO 객체로 매핑되어 List에 담깁니다.
	 */
	@Override
	public List<EmpDTO> selectAllEmp() {
		List<EmpDTO> resultList = null;
		
		resultList = sqlSession.selectList("mapper.emp.selectEmp");
		System.out.println("selectAllEmp: resultList: "+ resultList);
		
		return resultList;
	}

	/**
	 * [사원 단건 상세 조회]
	 * - sqlSession.selectOne(매퍼_식별자, 파라미터값)
	 *   : 조회 결과가 딱 1행(또는 0행)일 때 사용하는 메서드입니다.
	 *   : 두 번째 인자로 전달한 empno 값이 XML 매퍼의 #{_parameter} 위치에 바인딩됩니다.
	 */
	@Override
	public EmpDTO selectEmpno(int empno) {
		EmpDTO empDTO = null;
		
		empDTO = sqlSession.selectOne("mapper.emp.selectEmpno", empno);
		System.out.println("selectEmpno: empDTO: "+ empDTO);
		
		return empDTO;
	}

}
