package abstract_05;

import java.util.ArrayList;
import java.util.List;

/**
 * [추상 클래스 (Abstract Class)]
 * - 개념: "미완성 설계도" 또는 "공통 부모 전용 클래스"
 * - 특징:
 *   1) 내용이 덜 완성된 메서드(abstract 메서드)를 가질 수 있습니다.
 *   2) 미완성 설계도이므로 직접 물건(객체)을 찍어낼 수 없습니다. -> new Student() 불가능!
 *   3) 오직 다른 자식 클래스에게 '상속(extends)'되기 위해서만 존재합니다.
 */
public abstract class Student {
	
	// 학생이라면 공통으로 가질 수 있는 기술 목록 (필드/변수)
	List skills = new ArrayList();
	
	// [생성자]
	// 추상 클래스는 직접 new 할 수는 없지만,
	// 자식 클래스(Sangwoo 등)가 생성될 때 부모로서 먼저 호출되어 실행됩니다.
	Student() {
		System.out.println("Student 생성자 실행");
	}
	
	// [일반 메서드]
	// 모든 학생이 똑같이 행동하는 공통 기능은 부모가 미리 직접 만들어 둡니다.
	void eat() {
		System.out.println("우걱 우걱 밥 먹는다");
	}
	
	/**
	 * [추상 메서드 (Abstract Method)]
	 * - 개념: "이름만 있고 내용(실행 블록 {})은 없는 껍데기 메서드"
	 * - 특징:
	 *   1) 학생마다 공부하는 방식이 제각각이므로, 부모는 "공부하는 기능이 있어야 한다"는 틀만 잡습니다.
	 *   2) 자식 클래스에게 "너가 상속받아서 이 메서드의 실제 내용을 반드시 채워 넣어라!" 하고 강제(숙제)합니다.
	 *   3) 끝에 세미콜론(;)으로 끝납니다. (중괄호 {}가 있으면 에러 발생)
	 */
	abstract void study();
}
```

package interface_06;

/**
 * [인터페이스 (Interface)]
 * - 개념: "모든 기기에서 공통으로 지켜야 할 표준 규격(버튼 목록)"
 * - 특징:
 *   1) 변수를 선언하면 무조건 '상수(public static final)'가 됩니다. (값 변경 불가)
 *   2) 메서드를 선언하면 기본적으로 '추상 메서드(public abstract)'가 됩니다. (틀만 제공)
 *   3) 자바 8부터는 default 키워드를 붙여 기본 동작을 가진 메서드도 작성 가능합니다.
 */
public interface RemoteControl {
	// [인터페이스의 상수]
	// 기본적으로 'public static final' 속성을 갖기 때문에 생략해도 똑같이 상수가 됩니다.
	public static final int 보증기간 = 3;
	
	// 'public static final'이 생략되어 있어도 자동으로 상수로 취급됩니다.
	String company = "휴먼전자";
	
	// [추상 메서드]
	// 이 리모컨 규격을 따르는 기기(TV, 선풍기 등)는 무조건 전원 켜기 기능을 만들어야 합니다.
	public abstract void powerOn();
	
	// 'public abstract'가 생략되어 있어도 자동으로 추상 메서드로 취급됩니다.
	void powerOff();

	// 기존 기기들을 고치지 않고도 새로운 공통 기능을 추가할 때 유용하게 쓰입니다.

## EmpServiceImpl.java

package webservice_07;

import java.util.ArrayList;
import java.util.List;

/**
 * [서비스 구현 클래스 (Service Implementation)]
 * - 개념: "실제 비즈니스 로직(업무 규칙)을 수행하는 일꾼(요리사)"
 * - 흐름:
 *   1) DAO에게 부탁해 원본 데이터(재료)를 받아옵니다.
 *   2) 우리 회사의 규칙(예: 번호가 3000보다 큰 사원만 필터링)에 맞춰 데이터를 가공/선별합니다.
 *   3) 가공된 결과를 호출한 쪽(Controller)에 전달합니다.
 */
public class EmpServiceImpl implements EmpService {

	@Override 
	public List getEmp() {
		// 1. 데이터 창고 관리자(DAO)를 불러옴
		EmpDAO empDAO = new EmpDAOImpl();
		
		// 2. DB에서 원본 사원 목록 조회 (1111, 2222, 3333, 4444, 5555)
		List list = empDAO.SelectEmp();
		
		// 3. 비즈니스 로직(업무 규칙) 적용: 3000 초과인 사원만 골라 담기
		List result = new ArrayList();
		for(int i = 0; i < list.size(); i++) {
			if((int) list.get(i) > 3000) {
				result.add(list.get(i)); // 3333, 4444, 5555만 추가됨
			}
		}
		
		// 4. 필터링된 최종 목록 반환
		return result;
	}

}

package webservice_07;

import java.util.List;

/**
 * [JSP (JavaServer Pages / View 계층)]
 * - 개념: "사용자(브라우저)에게 보여줄 화면(HTML)을 예쁘게 플레이팅하는 역할"
 * - Controller로부터 순수 데이터(List)를 넘겨받아, 브라우저가 읽을 수 있는 HTML 태그 문자열로 조립합니다.
 */
public class Jsp {
	
	// 사원 번호 리스트를 받아 HTML 태그(<div>)로 감싸서 문자열로 만들어주는 메서드
	String view(List<Integer> list) {
		// 문자열을 빈번하게 합칠 때는 '+' 연산자보다 성능이 뛰어난 StringBuffer를 사용합니다.
		StringBuffer sb = new StringBuffer();
		
		// 향상된 for문(foreach)으로 리스트의 사원 번호를 하나씩 꺼내 HTML 태그로 조립
		for(int empno : list) {
			sb.append("<div>" + empno + "<div>\n");
		}
		
		// 조립이 끝난 StringBuffer를 일반 문자열(String)로 변환하여 반환
		String result = sb.toString();
		return result;
	}
}
```

		List<Integer> result = new ArrayList<Integer>();

			if((int) list.get(i) > 3000) {

		// 가공된 순수 데이터를 사용자가 브라우저에서 볼 수 있도록 HTML 문자열로 조립

package try_08;

import java.io.FileInputStream;
import java.io.IOException;

/**
 * [예외 처리 (Exception Handling)]
 * - 개념: "프로그램 실행 중 예기치 못한 사고(에러)가 터졌을 때 프로그램이 비정상 종료되지 않게 막아주는 에어백/안전장치"
 * - try: 에러가 발생할 가능성이 있는 코드 작성
 * - catch: 실제로 에러가 발생했을 때 수습(처리)하는 코드 작성
 * - finally: 에러가 나든 안 나든 무조건 마지막에 반드시 실행되는 뒷정리 코드 작성
 */
public class TryExam {

	// main 메서드 뒤의 'throws Exception'은 혹시 처리되지 않은 에러가 있다면 자바 가상머신(JVM)에 떠넘기겠다는 선언
	public static void main(String[] args) throws Exception {
		
		// -------------------------------------------------------------
		// [방식 1: 고전적인 try-catch-finally 자원 해제 방식]
		// -------------------------------------------------------------
		FileInputStream fis = null;
		
		try {
			// 파일 열기 시도 (만약 c:\test.txt 파일이 없으면 FileNotFoundException 발생)
			fis = new FileInputStream("c:\\test.txt");
			
		} catch(Exception e) {
			// 에러가 났을 때 빨간색으로 에러 추적 로그(어디서 왜 터졌는지)를 출력
			e.printStackTrace();
			
		} finally {
			// [뒷정리 영역]
			// 파일을 열었으면 컴퓨터 메모리와 OS 자원을 위해 반드시 닫아주어야(close) 합니다.
			// 에러가 나든 정상 실행되든 finally는 100% 실행됩니다.
			if(fis != null) {
				try {
					fis.close(); // 닫는 도중에도 에러가 날 수 있으므로 또 try-catch가 필요해서 코드가 지저분해짐
				} catch(IOException e) {
					e.printStackTrace();
				} catch(Exception e) {
					e.printStackTrace();
				}
			}
		}
		
		// -------------------------------------------------------------
		// [방식 2: 최신 자바의 try-with-resources 자동 자원 해제 문법 (Java 7+)]
		// -------------------------------------------------------------
		// try 뒤의 괄호() 안에 파일을 열어두면,
		// try 블록이 끝나는 순간 자바가 알아서 자동으로 close()를 호출해 줍니다! (finally 불필요, 매우 깔끔)
		try (
			FileInputStream fis2 = new FileInputStream("C:\\text.txt");
		) {
			System.out.println("뭔가 함");
		} catch (Exception e) {
			e.printStackTrace();
		}
		
		// -------------------------------------------------------------
		// [방식 3: 고의로 예외(에러)를 발생시키는 throw]
		// -------------------------------------------------------------
		// 특정 조건에서 개발자가 직접 의도적으로 에러를 터뜨려 흐름을 중단시킬 때 사용합니다.
		throw new Exception("내가 만든 에러");
		
	}

}

package string_09;

/**
 * [문자열(String) 다루기 실습]
 * - 자바에서 문자열을 검색하고, 자르고, 바꾸고, 검사하는 가장 핵심적인 기능 모음
 */
public class StringExam {

	public static void main(String[] args) {
		String s = "영일이삼사오육칠팔구삼사";
		
		// [1] 특정 위치(인덱스)의 글자 하나 가져오기 (0부터 시작: 0=영, 1=일, 2=이, 3=삼)
		char c = s.charAt(3);
		System.out.println(c); // 출력: 삼
		
		// [2] 문자열의 전체 글자 수(길이) 구하기
		System.out.println(s.length()); // 출력: 12
		
		// [3] 앞에서부터 특정 글자가 처음 등장하는 위치(인덱스) 찾기
		int i = s.indexOf("삼사");
		System.out.println(i); // 출력: 3 (3번째 위치에서 "삼사" 시작)
		
		// [4] 뒤에서부터 거꾸로 검색하여 마지막으로 등장하는 위치(인덱스) 찾기
		int i2 = s.lastIndexOf("삼사");
		System.out.println(i2); // 출력: 10 (10번째 위치의 "삼사")
		
		// -------------------------------------------------------------
		// [실습] 이메일 유효성 검사 2가지 방식
		// -------------------------------------------------------------
		String email = "tsunderechan@naver.com";
		
		// 방법 A: indexOf() 내장 메서드 사용
		// indexOf는 찾는 글자가 없으면 -1을 반환합니다.
		// 즉, @와 .의 위치가 모두 -1이 아니면 둘 다 존재한다는 뜻!
		if(email.indexOf("@") != -1 && email.indexOf(".") != -1 ) {
			System.out.println("이메일 맞음");
		} else {
			System.out.println("이메일 아님");
		}
		
		// 방법 B: indexOf를 쓰지 않고 for 반복문과 charAt()으로 직접 구현하기
		int cnt_골벵이 = -1; // @가 발견된 위치 저장 (발견 못 하면 -1 유지)
		int cnt_쩜 = -1;     // .이 발견된 위치 저장 (발견 못 하면 -1 유지)
		
		for(int j = 0; j < email.length(); j++) {
			if(email.charAt(j) == '@') {
				cnt_골벵이 = j; // @를 찾으면 해당 위치(인덱스) 기록
			} else if (email.charAt(j) == '.') {
				cnt_쩜 = j;    // .을 찾으면 해당 위치(인덱스) 기록
			}
		}
		
		// 둘 다 발견되었는지(-1이 아닌지) 확인
		if (cnt_골벵이 != -1 && cnt_쩜 != -1) {
			System.out.println("이메일 맞음 :" + cnt_골벵이 + "," + cnt_쩜);
	 	} else {
	 		System.out.println("이메일 아님");
	 	}
		
		// -------------------------------------------------------------
		// [5] 문자열 내용 바꾸기 (replace)
		// -------------------------------------------------------------
		// 주의: 원본 s는 바뀌지 않고, 변환된 새로운 문자열이 s2에 대입됩니다. (String의 불변성)
		String s2 = s.replace("삼사" , "34");
		System.out.println(s);  // 원본: 영일이삼사오육칠팔구삼사
		System.out.println(s2); // 변환본: 영일이34오육칠팔구34
		
		// -------------------------------------------------------------
		// [6] 문자열 일부분 잘라내기 (substring)
		// -------------------------------------------------------------
		// s.substring(시작, 끝): 시작 인덱스부터 '끝 인덱스 직전'까지 추출
		s2 = s.substring(5, 8); // 5번부터 7번까지 -> "오육칠"
		System.out.println(s2);
		
		// 시작 위치만 주면 끝까지 잘라냄
		s2 = s.substring(5);    // 5번부터 끝까지 -> "오육칠팔구삼사"
		System.out.println(s2);
		
		// -------------------------------------------------------------
		// [7] 특정 구분자를 기준으로 문자열 쪼개기 (split)
		// -------------------------------------------------------------
		// "삼" 글자를 기준으로 앞뒤로 쪼개어 배열(String[])에 담아줍니다.
		String[] s3 = s.split("삼");
		for (String a : s3) {
			System.out.println(a);
		}
		
		// -------------------------------------------------------------
		// [8] 0.0 이상 1.0 미만의 무작위 난수(실수) 생성 (Math.random)
		// -------------------------------------------------------------
		double r = Math.random();
		System.out.println(r);
	}

}
