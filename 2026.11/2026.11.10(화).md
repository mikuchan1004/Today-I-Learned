package melon_02;

import java.util.ArrayList;
import java.util.List;

/**
 * [Melon 클래스 - 음원 스트리밍 서비스 플랫폼]
 * 
 * 💡 핵심 개념:
 * 1. 컬렉션 프레임워크 (List & ArrayList):
 *    - 고정 크기의 일반 배열(Song[]) 대신, 필요할 때마다 곡을 무제한으로 담을 수 있는 '동적 리스트'를 사용합니다.
 *    - songs.add(곡): 곡 추가
 *    - songs.get(순번): 순번(인덱스)에 위치한 곡 가져오기
 *    - songs.size(): 현재 저장된 총 곡 수
 * 2. 방어적 프로그래밍(Defensive Coding):
 *    - 사용자가 저장된 곡 수보다 더 많은 수를 출력해달라고 요청해도,
 *      프로그램이 에러(IndexOutOfBoundsException)로 튕기지 않도록 안전장치를 두는 기법입니다.
 */
public class Melon {
	
	// ==========================================
	// 1. 필드(Field) : 멜론 서비스가 관리하는 데이터
	// ==========================================
//	Song[] songs = new Song[0]; // 일반 배열: 크기가 고정되어 데이터 추가/삭제가 매우 불편함
	List<Song> songs = new ArrayList<Song>(); // List: 크기가 자유자재로 늘어나는 편리한 리스트 바구니
	
	// ==========================================
	// 2. 생성자(Constructor) : 기본 생성자 사용 (별도 정의 없음)
	// ==========================================
	
	
	// ==========================================
	// 3. 메소드(Method) : 멜론의 주요 기능
	// ==========================================
	
	/**
	 * 목록 보여주기 
	 * 메소드명 : showList
	 * 전달인자 : int count 보여줄 개수
	 * 리턴타입 : void (반환값 없음)
	 * 
	 * songs의 내용 보여줄 count 만큼만 출력하기 
	 */
	void showList(int count) {
		// [방어 코딩 (Defensive Coding)]
		// 만약 보관 중인 곡이 2곡뿐인데 count로 10개를 보여달라고 하면
		// 존재하지 않는 인덱스를 조회하여 에러가 발생하므로,
		// count 값을 현재 저장된 최대 크기(songs.size())로 제한해줍니다.
		if (count > songs.size()) {
			count = songs.size();
		}
		
		// 0번 곡부터 요청한 count 개수까지만 반복하며 화면에 출력
		for (int i=0; i<count; i++) {
			Song song = songs.get(i); // i번째 인덱스의 곡 객체를 하나 꺼냄
			System.out.println("title :" + song.title);
			System.out.println("singer :" + song.singer);
		}
	}
	
	/**
	 * 곡 추가 
	 * 받은 Song을 songs에 추가
	 * 단, 제목과 가수가 같으면 추가하지 않는다. 
	 * 메소드명 : addSong
	 * 전달인자 : Song (추가하려는 음악 객체)
	 * 리턴타입 : 추가 성공 여부 (true: 성공, false: 실패)
	 */
	boolean addSong(Song song) {
		boolean result = false; // 추가 성공 여부를 저장할 변수 (기본값: 실패)
		
		// [케이스 1] 멜론 보관함이 완전히 비어있는 경우 (곡 개수가 0일 때)
		if(this.songs.size() == 0) {
			songs.add(song); // 중복 검사할 대상이 없으므로 바로 추가
			result = true;   // 성공으로 변경
		}else{
			// [케이스 2] 이미 보관함에 다른 곡들이 들어있는 경우
			// 기존 곡들을 하나씩 꺼내어 새로 들어온 song과 중복인지 검사
			for(int i=0; i<this.songs.size(); i++) {
				Song s = songs.get(i);
				
				// 제목(title)과 가수(singer)가 모두 일치하지 않는 경우(! 연산자)에만 추가
				if(!(s.title.equals(song.title) && s.singer.equals(song.singer))) {
					
					songs.add(song); // 목록에 새 곡 추가
					result = true;   // 성공으로 변경
				}
			}
		
		}
		
		return result; // 최종 성공/실패 여부를 호출한 곳으로 반환
	}
	
	/**
	 * 전곡 재생 기능
	 * 현재 멜론에 담겨있는 모든 곡의 play() 메서드를 순서대로 실행합니다.
	 */
	void playAll() {
		// 0번부터 마지막 곡(songs.size() - 1)까지 순회
		for (int i=0; i<this.songs.size(); i++) {
			songs.get(i).play(); // 각 Song 객체의 재생 메서드 호출
		}
	}
	
}

package melon_02;

/**
 * [MelonExam 실행 클래스 - 실제 음악 객체를 만들고 멜론 서비스를 구동하는 메인 테스트]

 * 💡 핵심 흐름:
 * 1. Melon 서비스 객체 생성 (스트리밍 플랫폼 준비)
 * 2. Song 객체 생성 (실제 음원 데이터 제작)
 * 3. melon.addSong(...)으로 음원 등록 및 결과 확인
 * 4. melon.showList(...)로 등록된 곡 목록 출력 (방어 코딩 동작 확인: 저장된 건 2곡인데 10곡 출력을 요청해도 에러 없음)

 */
public class MelonExam {

	public static void main(String[] args) {
		
		// 1. Melon 플랫폼 객체 생성 (메모리에 멜론 시스템 적재)
		Melon melon = new Melon();
		
		// 2. 첫 번째 곡 객체 생성: 제목 "With", 가수 "리센느"
		Song s1 = new Song("With" , "리센느");
		
		// 멜론에 s1 추가 시도 후 성공 여부(true/false)를 result 변수에 저장
		boolean result = melon.addSong(s1);
		if (result) {
			System.out.println("s1이 정상적으로 추가되었습니다.");
		}else {
			System.out.println("추가되지 않았습니다. 관리자 콜!");
		}
		
		// 3. 두 번째 곡 객체 생성: 제목 "꿈꾸던 어른이 되었나요?", 가수 "머무르"
		Song s2 = new Song("꿈꾸던 어른이 되었나요?" , "머무르");
		melon.addSong(s2); // 멜론 목록에 추가
		
		// 4. 저장된 곡 목록 10개 출력 요청
		// 💡 방어 코딩 검증: 현재 2곡만 등록되어 있으므로, 10개를 요청해도 2곡만 깔끔하게 출력됨 (에러 방지)
		melon.showList(10);
		
		// 5. 보관함에 등록된 모든 곡 일괄 재생
		melon.playAll();

	}

}

package melon_02;

/**
 * [Song 클래스 - 음악 1곡의 정보를 담는 설계도]
 * 
 * 💡 핵심 개념:
 * 1. 클래스(Class)와 객체(Object):
 *    - 클래스는 '음악'이라는 개념을 표현하기 위한 붕어빵 틀(설계도)입니다.
 *    - new Song(...)으로 실제 음악 데이터 1곡(객체/인스턴스)을 찍어냅니다.
 * 2. 접근 제어자(Access Modifier)와 캡슐화(Encapsulation):
 *    - private: 클래스 내부에서만 접근 가능하도록 감춥니다(중요한 데이터인 순위, 좋아요 수를 외부에서 임의 조작 방지).
 *    - default(생략): 같은 패키지(melon_02) 안에서 자유롭게 접근할 수 있습니다.
 */
public class Song {
	// ==========================================
	// 1. 필드(Field) : 음악 1곡이 가지는 고유한 속성(데이터)들
	// ==========================================
	// 제목, 가수명, 앨범명, 순위, 좋아요 수, 앨범아트
	String title;        // 곡 제목 (예: "With", "꿈꾸던 어른이 되었나요?")
	String singer;       // 가수 이름 (예: "리센느", "머무르")
	String album;        // 앨범 이름
	
	// private: 외부에서 마음대로 song.rank = 1; 처럼 조작할 수 없도록 안전하게 숨김(보호)
	private int rank;    // 음원 차트 순위
	private long like;   // 좋아요 수 (숫자가 매우 커질 수 있으므로 long 사용)
	
	String albumArtUrl;  // 앨범 재킷 이미지 파일 경로나 웹 주소
	boolean useYN;       // 음원 서비스 사용 가능 여부 (true: 재생가능, false: 서비스중지)
	
	// ==========================================
	// 2. 생성자(Constructor) : 객체를 처음 만들 때(new) 자동으로 실행되는 초기화 함수
	// ==========================================
	// 제목, 가수
	/**
	 * Song 객체 생성자
	 * @param title  곡 제목
	 * @param singer 가수명
	 * 
	 * 💡 this 키워드:
	 *  - 'this.title'은 이 클래스 자체의 필드 변수를 가리키고,
	 *  - 뒤의 'title'은 매개변수로 외부에서 넘겨받은 값을 뜻합니다.
	 */
	public Song(String title, String singer) {
		// 필수 정보는 외부에서 직접 전달받아 세팅
		this.title = title;
		this.singer = singer;
		
		// 그 외의 필드들은 기본 초기값으로 안전하게 세팅
		this.album = title;              // 앨범명을 따로 정하지 않으면 기본적으로 곡 제목과 동일하게 설정
		this.rank = -1;                  // 순위가 아직 매겨지지 않았음을 나타내는 임의의 초기값(-1)
		this.like = 0;                   // 좋아요 수는 처음엔 0개부터 시작
		this.albumArtUrl = "기본이미지";  // 앨범 이미지가 없을 때 띄울 기본 이미지 경로
		this.useYN = true;               // 기본적으로 이용 가능한 상태로 설정
	}
	
	// ==========================================
	// 3. 메소드(Method) : 음악 객체가 수행할 수 있는 동작/기능
	// ==========================================
	// 재생, 다운로드
	
	/**
	 * 음악 재생 기능
	 * 콘솔창에 현재 곡이 재생된다는 메시지를 출력합니다.
	 */
	public void play() {
		System.out.println(this.title + "을 재생합니다.");
	}
	
	/**
	 * 음원 다운로드 기능
	 * 다운로드할 수 있는 가상의 주소를 문자열로 만들어 돌려줍니다(return).
	 */
	String download() {
		return this.title + "다운로드 주소";
	}
	
	/**
	 * 좋아요 숫자를 변경한다
	 * - 좋아요를 눌렀을 때 1 증가 
	 * - 좋아요를 뺐을 때 1 감소 
	 *  
	 * @param boolean action
	 * 		true  : 좋아요를 눌렀을 때 (추천 +1)
	 * 		false : 좋아요를 취소했을 때 (추천 -1)
	 * @return long 변경된 최종 좋아요 수
	 * 
	 * 💡 private으로 감춰둔 like 변수를 안전하게 변경할 수 있도록 제공하는 비즈니스 메서드입니다.
	 */
	long updateLike(boolean action) {
		if(action) {
			this.like++;   // action이 true면 1 증가
		}else {
			this.like--;   // action이 false면 1 감소
		}
		
		return this.like;  // 계산이 끝난 최종 좋아요 수를 호출한 곳으로 반환
	}
}

## Phone.java

package Override_03;

import melon_02.Song; // melon_02 패키지에 만들어둔 Song 클래스를 벨소리로 활용하기 위해 불러옴(import)

/**
 * [Phone 클래스 - 기본 전화기 (부모 클래스 / Super Class)]
 * 
 * 💡 핵심 개념:
 * 1. 상속의 부모 클래스:
 *    - 일반 전화기(Phone)는 전화 걸기, 받기, 문자 보내기, 벨소리 울리기 등 가장 기본적인 통신 기능을 가집니다.
 *    - 나중에 스마트폰(SmartPhone)이 이 Phone 클래스를 상속(extends)받아 기능을 물려받게 됩니다.
 * 2. final 키워드:
 *    - 'final'이 붙은 변수는 한 번 값이 정해지면 절대 수정할 수 없는 상수(읽기 전용)가 됩니다.
 *    - 전화기 번호(number)는 개통 시 결정되면 기기 자체에서 임의로 바꿀 수 없도록 보호합니다.
 * 3. 캡슐화와 Getter / Setter:
 *    - private으로 보호된 변수(power, ring)는 외부에서 직접 접근할 수 없으므로,
 *      안전한 통로인 Getter(읽기)와 Setter(쓰기) 메서드를 통해 다룹니다.
 */
public class Phone {
	
	// ==========================================
	// 1. 필드(Field) : 전화기의 속성
	// ==========================================
	final String number;   // 내 전화번호 (final: 개통 후 번호 변경 불가)
	private boolean power; // 전원 상태 (true: 켜짐, false: 꺼짐) -> 외부 직접 변경 방지(private)
	private Song ring;     // 벨소리 (melon_02.Song 객체를 벨소리로 보관)
	
	// ==========================================
	// 2. 생성자(Constructor) : 전화기 개통 시 초기 상태 설정
	// ==========================================
	Phone(String number) {
		this.number = number;                          // 부여받은 전화번호 저장
		this.power = false;                            // 처음에는 전원이 꺼진 상태
		this.ring = new Song("기본벨" , "삼성");        // 초기 벨소리는 "삼성"의 "기본벨"로 장착
	}
	
	// ==========================================
	// 3. 메서드(Method) : 전화기의 통신 기능
	// ==========================================
	
	/**
	 * 전화 걸기 기능
	 * @param number 전화를 걸 상대방 번호
	 * @return true(통화 연결 성공), false(통화 연결 실패)
	 * 
	 * 💡 Math.random(): 0.0 ~ 1.0 사이의 난수를 생성하여 50% 확률로 통화 성공 여부를 시뮬레이션합니다.
	 */
	boolean call(String number) {
		if(Math.random() > 0.5) {
			System.out.println("전화 연결 성공 :" + number);
			return true;
		}else {
			System.out.println("전화 연결 실패 :" + number );
			return false;
		}
		
	} 
	
	/**
	 * 전화 받기 기능
	 */
	void receive() {
		System.out.println("누구세요?");
	}
	
	/**
	 * 문자 메시지 전송 기능
	 * @param number 받는 사람 번호
	 * @param text   보낼 내용
	 */
	void message(String number, String text) {
		System.out.println(number + "번호로" + text + "라고 보냈습니다.");
	}
	
	/**
	 * 전화벨 울리기 기능
	 * 현재 설정된 벨소리(Song) 객체의 play() 메서드를 호출하여 재생합니다.
	 */
	void ring() {
		// 전화벨 출력
		this.ring.play();
	}
	
	/**
	 * 벨소리 변경 기능
	 * @param song 새로 설정할 Song 객체
	 */
	void ringtone(Song song) {
		// 전화벨 교체
		this.ring = song;
	}

	// ==========================================
	// 4. Getter / Setter : 은닉된 private 필드를 안전하게 다루는 통로
	// ==========================================
	
	// 전원 켜짐/꺼짐 상태 확인
	public boolean isPower() {
		return power;
	}

	// 전원 켜기/끄기 설정
	public void setPower(boolean power) {
		this.power = power;
	}

	// 현재 설정된 벨소리 객체 조회
	public Song getRing() {
		return ring;
	}

	// 벨소리 교체
	public void setRing(Song ring) {
		this.ring = ring;
	}

	// 전화번호 조회 (final 변수이므로 setNumber는 없고 getNumber만 존재)
	public String getNumber() {
		return number;
	}	
	
}

package Override_03;

import melon_02.Song; // 벨소리로 사용할 Song 클래스 임포트

/**
 * [PhoneExam 실행 클래스 - 일반 전화기와 스마트폰의 동작 및 상속 관계 실습]

 * 💡 핵심 흐름:
 * 1. 일반 Phone 객체(phone1, phone2) 생성 및 통화 시뮬레이션
 * 2. 외부 Song 객체를 생성하여 phone1의 벨소리 교체 (ringtone)

package Override_03;

/**
 * [SmartPhone 클래스 - 스마트폰 (자식 클래스 / Sub Class)]
 * 
 * 💡 핵심 개념:
 * 1. 상속 (extends):
 *    - 'SmartPhone extends Phone'은 스마트폰이 일반 전화기(Phone)의 모든 기능(전화 걸기/받기/문자 등)을 물려받는다는 의미입니다.
 *    - 자바는 다중 상속을 금지하므로 '단 하나의 부모 클래스'만 상속받을 수 있습니다.
 *    - 부모의 private 요소는 은닉되어 있으므로 자식이라도 직접 접근할 수 없습니다.
 * 2. super 키워드:
 *    - super(number): 부모 클래스(Phone)의 생성자를 호출합니다. 자식 객체가 만들어질 때는 반드시 부모 생성자가 먼저 실행되어야 합니다.
 *    - super.메서드(): 부모가 가지고 있던 원래 메서드를 명시적으로 호출할 때 사용합니다.
 * 3. 메서드 오버라이딩 (Method Overriding / @Override):
 *    - 부모로부터 물려받은 메서드를 그대로 쓰지 않고, 자식 클래스에서 내용을 새롭게 '재정의'하여 덮어쓰는 기법입니다.
 *    - @Override 어노테이션은 부모의 메서드 시그니처와 정확히 일치하는지 컴파일러에게 검사해달라고 알리는 역할을 합니다.
 */
public class SmartPhone extends Phone {
	
	// ==========================================
	// 1. 생성자(Constructor) : 스마트폰 초기화
	// ==========================================
	SmartPhone(String number) {
		// [super(number)]
		// 부모 클래스(Phone)의 생성자 Phone(String number)를 호출하여 부모 영역을 먼저 초기화합니다.
		// 자식 생성자 맨 첫 줄에는 반드시 super(...)가 와야 합니다.
		super(number);
		
		// 부모에게 물려받은 필드(number)와 메서드(ring())를 내 것처럼 자유롭게 사용 가능
		System.out.println(this.number);
		ring();
		
		// 💡 주의: 부모의 private 변수(power)는 자식이라도 직접 꺼내 쓸 수 없습니다 (컴파일 에러 발생하므로 주석 처리됨).
		// 접근하려면 부모의 public 메서드인 isPower()나 setPower()를 거쳐야 합니다.
//		System.out.println(this.power);
	}
	
	// ==========================================
	// 2. 스마트폰만의 고유한 기능 추가 (확장)
	// ==========================================
	
	/**
	 * 앱 스토어 열기 기능 (스마트폰에만 있는 고유 기능)
	 */
	void openStore() {
		System.out.println("어플 목록이 나옵니다.");
	}
	
	/**
	 * 웹 검색 기능 (스마트폰에만 있는 고유 기능)
	 * @param keyword 검색어
	 * @return 가상의 검색 결과 웹페이지
	 */
	String search(String keyword) {
		System.out.println(keyword + "검색 중 입니다...");
		
		// 부모(Phone)의 원래 message 메서드를 호출하여 로그나 알림을 보냄
		super.message("1" , "text");
		
		return keyword + "검색 결과 페이지";
	}
	
	// ==========================================
	// 3. 부모 기능 재정의 (메서드 오버라이딩 / @Override)
	// ==========================================
	
	/**
	 * [메서드 오버라이딩]
	 * 부모(Phone)의 message 메서드를 스마트폰 방식(예: 메신저 앱 스타일)으로 덮어써서 재정의합니다.
	 */
	@Override
	void message(String number, String text) {
		// super.message()를 통해 부모(Phone)의 기존 문자 전송 로직도 함께 수행
		super.message("1" , "text");
		
		// 스마트폰만의 추가적인 동작(출력) 수행
		System.out.println("1" + text);
	}
}

package game_04;

/**
 * [Auto 클래스 - 챔피언 매크로 자동 실행 시스템 (다형성과 다운캐스팅의 핵심)]
 * 
 * 💡 핵심 개념:
 * 1. 매개변수의 다형성 (Polymorphism):
 *    - run(Champ champ): 매개변수의 타입이 부모인 'Champ'입니다.
 *    - 부모 타입 그릇 하나만 만들어 두면, 자식 객체인 Jax든 VI든 어떤 챔피언이 들어와도 다 받아낼 수 있습니다!
 *    - (만약 다형성이 없다면 run(Jax jax), run(VI vi) 등 챔피언 종류마다 메서드를 수십 개씩 따로 만들어야 합니다.)
 * 
 * 2. 동적 바인딩 (Dynamic Method Binding):
 *    - 'champ.move()'를 실행할 때, 겉보기 타입은 부모인 Champ이지만,
 *      자바는 메모리에 실제로 들어있는 진짜 알맹이(Jax인지 VI인지)를 스스로 찾아가서
 *      오버라이딩된 각자의 전용 move() 메서드를 똑똑하게 실행해줍니다.
 * 
 * 3. instanceof 연산자와 다운캐스팅 (Downcasting):
 *    - 부모 타입(Champ) 상태에서는 바이(VI)만의 고유 기능인 '징크스랑놀기()'가 보이지 않습니다.
 *    - 따라서 'champ instanceof VI'로 "너 진짜 바이 맞니?"라고 먼저 안전하게 확인한 뒤,
 *    - '(VI)champ'처럼 자식 타입으로 형태를 되돌려(다운캐스팅) 고유 기능을 실행합니다.
 *    - ※ 잭스(Jax)에게 억지로 바이 행세를 시키면 에러(ClassCastException)가 나므로 instanceof 검사가 필수입니다.
 */
public class Auto {
	
	/**
	 * 어떤 챔피언이든 전달받아서 공격, 이동, 스킬, 귀환을 자동으로 순차 실행하는 매크로 메서드
	 * @param champ 챔피언 객체 (Champ를 상속받은 모든 자식 챔피언 가능)
	 */
	void run(Champ champ) {
		// [공통 동작 실행]
		// 부모 Champ 규격에 있는 동작들이 차례대로 실행됩니다.
		champ.attack();    // 기본 공격
		champ.move();      // 💡 동적 바인딩: 들어온 챔피언(Jax/VI)의 오버라이딩된 대사가 각각 출력됨!
		champ.skill("F");  // 점멸 사용
		champ.skill("R");  // 궁극기 사용
		champ.recall();    // 귀환
		
		// [특정 챔피언 전용 고유 동작 처리 (다운캐스팅)]
		// 만약 지금 들어온 champ 객체의 실제 정체가 'VI(바이)'라면?
		if(champ instanceof VI) {			
			// Champ 타입 그릇을 VI 타입 그릇으로 안전하게 형변환(다운캐스팅)
			VI vi = (VI)champ;
			
			// 이제 바이만의 고유 메서드를 호출할 수 있게 됩니다!
			vi.징크스랑놀기();
		}
	}
}

 */

	public static void main(String[] args) {
		
		// ==========================================
		// 1. 자식 객체 직접 생성 (바이, 잭스)
		// ==========================================
		VI 바이 = new VI();

		// ==========================================
		// champ1은 현재 부모인 Champ 타입이므로 'champ1.징크스랑놀기()'를 바로 호출할 수 없습니다.

 * [VI 클래스 - 챔피언 '바이' (Champ를 상속받은 자식 클래스)]

 * 💡 핵심 개념:

 * 2. 메서드 오버라이딩 (@Override):
 *    - 부모의 move() 메서드를 바이의 고유한 이동 대사로 재정의했습니다.
 * 3. 자식 클래스만의 고유 메서드 (징크스랑놀기):
 *    - 부모(Champ)에게는 없고 오직 바이(VI)에게만 있는 특별한 기능입니다.
 *    - 💡 중요: 만약 'Champ champ = new VI();' 처럼 부모 타입으로 바이 객체를 참조하면,
 *      부모 규격에 '징크스랑놀기'가 정의되어 있지 않아 바로 호출할 수 없습니다.
 *      이때 원래 바이 타입으로 되돌리는 '다운캐스팅((VI)champ)'이 필요합니다!
 */
public class VI extends Champ {
	
	/**
	 * [오버라이딩] 바이만의 이동 동작
	 * 부모의 기본 이동 메서드를 덮어써서 실행됩니다.
	 */
	@Override
	void move() {
		System.out.println("앗흥 이동합니다. 앗흥");
	}
	
	/**
	 * 바이만의 고유 메서드 (부모 Champ에는 없는 특별한 기능)
	 */
	void 징크스랑놀기() {
		System.out.println("징크스야~ 놀~~~~자");
	}
}
