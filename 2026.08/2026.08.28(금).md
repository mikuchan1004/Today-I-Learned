### cookie.html

``` html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>JS 쿠키 제어</title>
    <style>
        /* 팝업창 기본 스타일 정의 */
        .popup {
            border : 2px solid black;
            padding : 20px;
            position: absolute;
            top : 20px;
            left : 30px;
            background-color : white;
        }
        /* 팝업을 화면에서 숨겨버리는 클래스 */
        .hide {
            display: none;
        }
    </style>
    <script>
        // 웹페이지의 모든 HTML 요소가 화면에 다 그려진(로드된) 직후 실행
        window.onload = () => {

            // [버튼 1 클릭] 만료일이 없는 '세션 쿠키' 만들기
            document.querySelector('#btn1').addEventListener('click' , () => {
                // max-age나 expires를 안 적으면 브라우저를 닫을 때 자동으로 삭제되는 임시 쿠키가 됩니다.
                document.cookie = "key=value";
            });

            // [버튼 2 클릭] 수명이 정해진 쿠키 만들기
            document.querySelector('#btn2').addEventListener('click' , () => {
                // max-age는 쿠키의 유효 수명(초 단위)입니다.
                // 60초 * 60분 * 2 = 7,200초 (2시간 동안 유지)
                document.cookie = `mode=attack; max-age=${60 * 60 * 2}`;
                
                // 60초 * 60분 * 24 = 86,400초 (24시간 = 1일 동안 유지)
                document.cookie = `heal=full; max-age=${60 * 60 * 24}`;
            });

            // [버튼 3 클릭] 쿠키 문자열을 직접 자르고 해석(파싱)하기
            document.querySelector('#btn3').addEventListener('click' , () => {
                // document.cookie는 "key=value; mode=attack; heal=full"처럼 모든 쿠키가 한 줄로 묶여 나옵니다.
                console.log(document.cookie);

                // [중요/수정] 쿠키 구분자인 세미콜론과 공백('; ')을 기준으로 문장을 싹 자릅니다.
                const cookies = document.cookie.split('; '); 

                // 자른 쿠키 조각들을 하나씩 돌려보면서 확인합니다.
                for (let cookie of cookies) {
                    console.log(cookie); // 예: "heal=full"

                    // 등호('=') 기준으로 이름과 값을 쪼갭니다.
                    let name = cookie.split('=')[0];  // 쿠키 이름 (예: heal)
                    let value = cookie.split('=')[1]; // 쿠키 값 (예: full)

                    // 쿠키 이름이 'heal'일 때 처리
                    if (name === 'heal') {
                        if (value === 'full') {
                            console.log('힐 합니다.');
                        } else {
                            console.log('힐 안합니다.');
                        }
                    }
                }
            });

            // ====================================================
            // 팝업 제어 로직 (페이지 접속하자마자 제일 먼저 실행)
            // ====================================================

            // 1. 페이지 접속 시 'popup=closed' 쿠키가 남아있는지 확인
            if (document.cookie.includes('popup=closed')) {
                // 쿠키가 존재하면 팝업에 hide 클래스를 붙여서 안 보이게 만듭니다.
                document.querySelector('.popup').classList.add('hide');
            }

            // 2. 닫기 버튼을 눌렀을 때의 동작
            document.querySelector('.close').addEventListener('click', () => {
                // 버튼을 누르면 일단 화면에서 팝업을 숨깁니다.
                document.querySelector('.popup').classList.add('hide');

                // [중요/수정] 체크박스가 '체크되어 있을 때만' 10초짜리 쿠키를 새로 생성합니다.
                if (document.querySelector('.chk').checked) {
                    document.cookie = `popup=closed; max-age=10`;
                }
            });
        }
    </script>
</head>
<body>
    <button type="button" id="btn1">btn1 (세션쿠키)</button>
    <button type="button" id="btn2">btn2 (수명쿠키)</button>
    <button type="button" id="btn3">btn3 (쿠키조회)</button>
    
    <div class="popup">
        <div>
            너무 너무 중요한 공지<br><br>
        </div>
        <div>
            <input type="checkbox" class="chk"> 10초간 안 보이기
            <button type="button" class="close">닫기</button>
        </div>
    </div>
</body>
</html>
```

## 파이썬으로 쿠키 다루기 

### cookie.py

``` python
from fastapi import FastAPI, Cookie, Request, Response
from fastapi.templating import Jinja2Templates 
from typing import Annotated

app = FastAPI()
# HTML 템플릿 파일들이 저장되어 있는 폴더 지정
templates = Jinja2Templates(directory='templates/') 

# 브라우저가 /main 주소로 접속했을 때 실행되는 함수
@app.get('/main')
def main(
     request: Request,
     response : Response,

     # [쿠키 읽기 1] 브라우저가 보낸 쿠키 중 'no'라는 이름을 찾아 알아서 변수에 저장합니다.
     # 쿠키가 없으면 기본값으로 None이 들어갑니다.
     no: str | None = Cookie(None),
     
     # [쿠키 읽기 2] 위와 동일한 기능인데, Annotated라는 최신 파이썬 문법 형태로 쓴 것입니다.
     yes : Annotated[str | None, Cookie()] = None 
     ):
     
     # 서버 터미널 창에 받은 쿠키 값을 출력해서 확인합니다.
     print('no :' , no)
     print('yes :' , yes)

     # 브라우저에 보여줄 HTML 화면(main.html) 응답 객체를 만듭니다.
     response = templates.TemplateResponse(request, 'main.html')

     # [쿠키 쓰기 1] 기본 세션 쿠키 설정 (브라우저 종료 전까지 유지)
     response.set_cookie(
         key = 'key',
         value = 'value'
     )

     # [쿠키 쓰기 2] 10초 후 자동 삭제되는 타이머 쿠키 설정
     response.set_cookie(
         key = 'key2',
         value = 'value2',
         max_age = 10 # 10초 수명
     )

     # [쿠키 쓰기 3] 보안 강화 쿠키 설정 (httponly=True)
     response.set_cookie(
         key = 'key3',
         value = 'value3',
         max_age = 1000,
         httponly = True # 자바스크립트(document.cookie)로 접근 불가능하게 막아 해킹 방지
     )

     # 준비된 응답(HTML + 쿠키 3개)을 브라우저로 보냅니다.
     return response


# 브라우저가 /delete/cookie 주소로 접속했을 때 쿠키 삭제 처리
@app.get('/delete/cookie')
def delete_cookie(response: Response):
     # 브라우저에게 'key3' 쿠키를 즉시 삭제하라고 명령합니다.
     response.delete_cookie('key3')
     return '{"message" : "쿠키 key3 삭제 완료"}'


if __name__ == "__main__":
    import uvicorn
    # 서버 실행 명령어 (포트 8000번)
    uvicorn.run("cookie:app" , port=8000, reload=True, host="0.0.0.0")
```

## 세션 연습 

### session.py

``` python
# pip install itsdangerous

from fastapi import FastAPI,  Request,  Response
from starlette.middleware.sessions import SessionMiddleware
from fastapi.templating import Jinja2Templates 
from fastapi.responses import RedirectResponse   

app = FastAPI()

templates = Jinja2Templates(directory='templates/')  

app.add_middleware(
    SessionMiddleware,
      secret_key='human1234$'
 )

@app.get('/login')
def login(request:Request):
    # 세션 저장
    request.session['isLogin'] = True
    request.session['id'] = "admin"
    return "로그인 성공!"

@app.get('/mypage')
def mypage(request:Request):
    isLogin = request.session.get('isLogin')
    id = request.session.get('id')
    if isLogin is None  :
        return "로그인 하세요"  
    else :
        return f" id: [{id}] 비밀스러운 공간에 오신 걸 환영합니다."

@app.get('/logout')
def logout(request:Request):
    # invalidate
    request.session.clear()
    return "로그아웃"

@app.get('/')
def home(request:Request):
    return templates.TemplateResponse(request, 'main.html')

if __name__ == "__main__":
    import uvicorn
    uvicorn.run("session:app" , port=8000, reload=True, host="0.0.0.0")
```
